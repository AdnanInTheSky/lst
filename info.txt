  <script>
    const API_BASE = "/api";
    const params = new URLSearchParams(location.search);
    const invoice = params.get('invoice_number') || '';
    let lastOrder = null;
    try { lastOrder = JSON.parse(sessionStorage.getItem('lastOrder')); } catch {}

    function setText(id, text) {
      const el = document.getElementById(id);
      if (el) el.textContent = text;
    }

    async function verify(attempt = 1) {
      if (!invoice) {
        render('failed','❌ কোনো ইনভয়েস নেই','URL-এ ইনভয়েস নম্বর অনুপস্থিত।');
        return;
      }
      try {
        const res = await fetch(API_BASE + '/status', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ invoice_number: invoice })
        });
        const data = await res.json();
        const status = (data.trx_status || data.status || '').toLowerCase();
        if (status === 'successful' || status === 'success') {
          render('success','🎉 অর্ডার নিশ্চিত!','আপনার পেমেন্ট সফল হয়েছে। আমবাড়ি পরিবারের পক্ষ থেকে ধন্যবাদ!', data);
        } else if (status === 'failed' || status === 'cancelled') {
          render('failed','❌ পেমেন্ট ব্যর্থ','আপনার পেমেন্ট সম্পন্ন হয়নি। আপনার অ্যাকাউন্ট থেকে কোনো টাকা কাটা হয়নি।', data);
        } else if (attempt < 12) {
          setTimeout(() => verify(attempt + 1), 5000);
        } else {
          render('failed','❌ যাচাই করা যায়নি','অনুগ্রহ করে সাপোর্টের সাথে যোগাযোগ করুন।', data);
        }
      } catch(e) {
        render('failed','❌ নেটওয়ার্ক ত্রুটি','ইন্টারনেট সংযোগ চেক করে আবার চেষ্টা করুন।');
      }
    }

    function render(type, title, sub, data = {}) {
      const icon = document.getElementById('status-icon');
      icon.className = 'w-20 h-20 rounded-full flex items-center justify-center text-4xl mx-auto mb-5 ' +
        (type === 'success'
          ? 'bg-gradient-to-br from-[#e8f5ee] to-[#d4edda]'
          : type === 'failed'
            ? 'bg-gradient-to-br from-[#fde8e6] to-[#f5c6cb]'
            : 'animate-pulse-ring bg-gradient-to-br from-[#fdf5e0] to-cream');
      icon.textContent = type === 'success' ? '✅' : '❌';
      setText('status-title', title);
      setText('status-sub', sub);

      if (invoice) {
        setText('inv-number', invoice);
        setText('inv-name', lastOrder?.name || data?.customer?.name || '—');
        setText('inv-amount', lastOrder?.amount ? '৳' + Number(lastOrder.amount).toLocaleString('bn-BD') + ' BDT' : '—');
        setText('inv-status', data?.trx_status || type);
        setText('inv-trxid', data?.trx_id || '—');
        document.getElementById('invoice-section').classList.remove('hidden');
      }

      if (lastOrder?.items?.length) {
        document.getElementById('items-list').innerHTML = lastOrder.items.map(it =>
          `<div class="flex justify-between py-2 border-b border-brand-border text-sm last:border-b-0">
            <div class="flex items-center gap-2.5">
              <img src="${escapeAttr(it.image)}" alt="${escapeAttr(it.name)}" class="w-10 h-10 rounded-lg object-cover" />
              <span>${escapeHtml(it.name)} × ${it.qty}</span>
            </div>
            <span class="font-medium">৳${(it.price * it.qty).toLocaleString('bn-BD')}</span>
          </div>`
        ).join('');
        document.getElementById('items-section').classList.remove('hidden');
        sessionStorage.removeItem('lastOrder');
      }

      document.getElementById('card-footer').classList.remove('hidden');
      if (type === 'failed') setText('cta-btn', '← আবার চেষ্টা করুন');
      setText('verify-note', data?.trx_id ? 'ট্রানজেকশন আইডি: ' + data.trx_id : 'PayStation নিরাপদ গেটওয়ের মাধ্যমে যাচাইকৃত।');
    }

    function escapeHtml(str) {
      if (typeof str !== 'string') return '';
      return str.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
    }
    function escapeAttr(str) {
      if (typeof str !== 'string') return '';
      return str.replace(/"/g,'&quot;').replace(/'/g,'&#x27;');
    }

    function downloadPDF() {
      const element = document.getElementById('invoice-card');
      const opt = {
        margin:       [10, 10, 10, 10],
        filename:     `invoice-${invoice || 'order'}.pdf`,
        image:        { type: 'jpeg', quality: 0.98 },
        html2canvas:  { scale: 2, useCORS: true, logging: false },
        jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }
      };

      // Temporarily hide buttons for clean PDF
      const footer = document.getElementById('card-footer');
      const originalDisplay = footer.style.display;
      footer.style.display = 'none';

      html2pdf().set(opt).from(element).save().then(() => {
        footer.style.display = originalDisplay;
      }).catch(() => {
        footer.style.display = originalDisplay;
        alert('PDF তৈরি করতে সমস্যা হয়েছে। আবার চেষ্টা করুন।');
      });
    }

    verify();
  </script>
